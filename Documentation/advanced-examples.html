SwiftUI Integration

import SwiftUI
import BICurlDownloader

class DownloadViewModel: ObservableObject {
    @Published var downloads: [DownloadItem] = []
    @Published var totalProgress: Double = 0
    
    private let manager = BIDownloadManager.shared
    
    func startDownload(url: URL, filename: String) {
        let destinationPath = NSTemporaryDirectory() + filename
        let task = manager.download(url: url, destinationPath: destinationPath, note: filename)
        task.delegate = self
        
        let item = DownloadItem(task: task)
        downloads.append(item)
    }
}

extension DownloadViewModel: BIDownloadDelegate {
    func downloadDidProgress(_ task: BIDownloadTask, progress: Double) {
        DispatchQueue.main.async {
            if let index = self.downloads.firstIndex(where: { $0.task.id == task.id }) {
                self.downloads[index].progress = progress
            }
        }
    }
}

Network-Aware Download Manager

import Network

class SmartDownloadManager {
    private let pathMonitor = NWPathMonitor()
    private var currentPath: NWPath?
    
    func startDownload(url: URL) {
        let streamCount = getOptimalStreams()
        var options = BIDownloadOptions()
        options.maxConcurrentStreams = streamCount
        
        let task = BIDownloadManager.shared.download(url: url, destinationPath: path, options: options)
    }
    
    private func getOptimalStreams() -> Int {
        guard let path = currentPath else { return 4 }
        return path.usesInterfaceType(.wifi) ? 8 : 2
    }
}

Priority Download Queue

class PriorityDownloadQueue {
    enum Priority: Int {
        case high = 0, normal = 1, low = 2
    }
    
    private var queues: [Priority: [BIDownloadTask]] = [:]
    
    func addDownload(url: URL, priority: Priority) {
        let task = BIDownloadManager.shared.download(url: url, destinationPath: path)
        queues[priority, default: []].append(task)
    }
}
