Error Handling

Robust error handling and recovery strategies

Error Types

BICurlDownloader provides comprehensive error reporting through the BIDownloadError class.

Error Codes

Code Name Description Recovery
1001 invalidURL Invalid or malformed URL Fix URL format
1002 networkError Network connection issue Retry after delay
1003 fileSystemError File system operation failed Check permissions, disk space
1004 curlError cURL library error Check logs, retry
1005 audioSessionError Audio session configuration failed Check background capability
1006 validationError Invalid parameters or state Fix input parameters
1007 authenticationError Authentication failed Check credentials, refresh token
1008 serverError Server returned error status Check server status, retry
1009 rangeNotSupported Server doesn't support Range requests Disable multipart
1010 insufficientSpace Not enough disk space Free up space

Basic Error Handling

Swift Implementation

extension ViewController: BIDownloadDelegate {
    func downloadDidFail(_ task: BIDownloadTask, error: Error) {
        let nsError = error as NSError
        
        switch nsError.code {
        case BIDownloadErrorCode.invalidURL.rawValue:
            showAlert(title: "Invalid URL", message: "Please check the download URL")
            
        case BIDownloadErrorCode.networkError.rawValue:
            showAlert(title: "Network Error", message: "Check your internet connection")
            retryDownloadAfterDelay(task)
            
        case BIDownloadErrorCode.fileSystemError.rawValue:
            showAlert(title: "Storage Error", message: "Unable to save file")
            checkDiskSpace()
            
        case BIDownloadErrorCode.authenticationError.rawValue:
            showAlert(title: "Authentication Failed", message: "Please login again")
            requestNewCredentials()
            
        case BIDownloadErrorCode.insufficientSpace.rawValue:
            showAlert(title: "Storage Full", message: "Free up some space and try again")
            
        default:
            showAlert(title: "Download Failed", message: error.localizedDescription)
        }
    }
}

Automatic Retry Mechanism

Exponential Backoff Strategy

// Swift
class RetryManager {
    private var retryAttempts: [String: Int] = [:]
    private let maxRetries = 3
    
    func shouldRetry(taskId: String) -> Bool {
        let attempts = retryAttempts[taskId, default: 0]
        return attempts < maxRetries
    }
    
    func getRetryDelay(taskId: String) -> TimeInterval {
        let attempts = retryAttempts[taskId, default: 0]
        // Exponential backoff: 2^attempts seconds
        return pow(2.0, Double(attempts))
    }
    
    func recordRetry(taskId: String) {
        retryAttempts[taskId, default: 0] += 1
    }
    
    func resetRetries(taskId: String) {
        retryAttempts.removeValue(forKey: taskId)
    }
}

// Usage
let retryManager = RetryManager()

func downloadDidFail(_ task: BIDownloadTask, error: Error) {
    let nsError = error as NSError
    
    // Only retry network errors
    guard nsError.code == BIDownloadErrorCode.networkError.rawValue else {
        print("Non-retryable error: \(error)")
        return
    }
    
    if retryManager.shouldRetry(taskId: task.id) {
        let delay = retryManager.getRetryDelay(taskId: task.id)
        retryManager.recordRetry(taskId: task.id)
        
        print("Retrying in \(delay) seconds (attempt \(retryManager.retryAttempts[task.id] ?? 0)/\(retryManager.maxRetries))")
        
        DispatchQueue.main.asyncAfter(deadline: .now() + delay) {
            self.retryDownload(task)
        }
    } else {
        print("Max retries exceeded for task \(task.id)")
        retryManager.resetRetries(taskId: task.id)
        showFinalError(task, error: error)
    }
}

func retryDownload(_ task: BIDownloadTask) {
    // Create new download with same parameters
    let newTask = BIDownloadManager.shared.download(
        url: task.url,
        destinationPath: task.destinationPath,
        options: task.options
    )
    newTask.delegate = self
}

func downloadDidComplete(_ task: BIDownloadTask, filePath: String) {
    // Reset retry counter on success
    retryManager.resetRetries(taskId: task.id)
    print("Download completed: \(filePath)")
}

Objective-C Implementation

@interface RetryManager : NSObject
@property (nonatomic, strong) NSMutableDictionary *retryAttempts;
@property (nonatomic, assign) NSInteger maxRetries;
@end

@implementation RetryManager

- (instancetype)init {
    self = [super init];
    if (self) {
        _retryAttempts = [NSMutableDictionary dictionary];
        _maxRetries = 3;
    }
    return self;
}

- (BOOL)shouldRetryForTaskId:(NSString *)taskId {
    NSInteger attempts = [self.retryAttempts[taskId] integerValue];
    return attempts < self.maxRetries;
}

- (NSTimeInterval)getRetryDelayForTaskId:(NSString *)taskId {
    NSInteger attempts = [self.retryAttempts[taskId] integerValue];
    return pow(2.0, attempts);
}

- (void)recordRetryForTaskId:(NSString *)taskId {
    NSInteger attempts = [self.retryAttempts[taskId] integerValue];
    self.retryAttempts[taskId] = @(attempts + 1);
}

- (void)resetRetriesForTaskId:(NSString *)taskId {
    [self.retryAttempts removeObjectForKey:taskId];
}

@end

Network Error Handling

Common Network Issues

func handleNetworkError(_ error: Error, task: BIDownloadTask) {
    let nsError = error as NSError
    
    if let underlyingError = nsError.userInfo[NSUnderlyingErrorKey] as? NSError {
        switch underlyingError.code {
        case NSURLErrorNotConnectedToInternet:
            print("No internet connection")
            // Wait for network and retry
            monitorNetworkAndRetry(task)
            
        case NSURLErrorTimedOut:
            print("Request timed out")
            // Retry immediately
            retryDownload(task)
            
        case NSURLErrorCannotFindHost:
            print("Cannot find host")
            // Don't retry, show error
            showError("Server not found")
            
        case NSURLErrorNetworkConnectionLost:
            print("Connection lost")
            // Retry after delay
            retryAfterDelay(task, delay: 5.0)
            
        default:
            print("Network error: \(underlyingError.localizedDescription)")
        }
    }
}

func monitorNetworkAndRetry(_ task: BIDownloadTask) {
    import Network
    
    let monitor = NWPathMonitor()
    monitor.pathUpdateHandler = { [weak self] path in
        if path.status == .satisfied {
            print("Network available, retrying download")
            monitor.cancel()
            self?.retryDownload(task)
        }
    }
    monitor.start(queue: DispatchQueue.global())
}

Disk Space Management

Check Available Space

// Swift
func checkDiskSpace() -> (available: Int64, required: Int64)? {
    guard let path = NSSearchPathForDirectoriesInDomains(.documentDirectory, .userDomainMask, true).first else {
        return nil
    }
    
    do {
        let attributes = try FileManager.default.attributesOfFileSystem(forPath: path)
        
        if let freeSpace = attributes[.systemFreeSize] as? Int64 {
            let requiredSpace: Int64 = 100 * 1024 * 1024 // 100 MB buffer
            return (available: freeSpace, required: requiredSpace)
        }
    } catch {
        print("Error checking disk space: \(error)")
    }
    
    return nil
}

func downloadDidFail(_ task: BIDownloadTask, error: Error) {
    let nsError = error as NSError
    
    if nsError.code == BIDownloadErrorCode.insufficientSpace.rawValue {
        if let space = checkDiskSpace() {
            let availableMB = space.available / 1024 / 1024
            let requiredMB = space.required / 1024 / 1024
            
            let message = """
            Not enough storage space available.
            Available: \(availableMB) MB
            Required: \(requiredMB) MB
            
            Please free up some space and try again.
            """
            
            showAlert(title: "Storage Full", message: message)
        }
    }
}

Objective-C Implementation

- (void)checkDiskSpace {
    NSArray *paths = NSSearchPathForDirectoriesInDomains(NSDocumentDirectory, NSUserDomainMask, YES);
    NSDictionary *attributes = [[NSFileManager defaultManager] attributesOfFileSystemForPath:paths.firstObject error:nil];
    
    NSNumber *freeSpace = attributes[NSFileSystemFreeSize];
    int64_t availableBytes = [freeSpace longLongValue];
    int64_t availableMB = availableBytes / 1024 / 1024;
    
    NSLog(@"Available space: %lld MB", availableMB);
    
    if (availableMB < 100) {
        [self showStorageWarning];
    }
}

User Feedback

Progress Indicators

// Swift
class DownloadViewController: UIViewController {
    @IBOutlet weak var progressView: UIProgressView!
    @IBOutlet weak var statusLabel: UILabel!
    @IBOutlet weak var errorLabel: UILabel!
    
    func downloadDidStart(_ task: BIDownloadTask) {
        DispatchQueue.main.async {
            self.progressView.isHidden = false
            self.statusLabel.text = "Downloading..."
            self.errorLabel.isHidden = true
        }
    }
    
    func downloadDidProgress(_ task: BIDownloadTask, progress: Double) {
        DispatchQueue.main.async {
            self.progressView.progress = Float(progress)
            let percentage = Int(progress * 100)
            self.statusLabel.text = "Downloading: \(percentage)%"
        }
    }
    
    func downloadDidFail(_ task: BIDownloadTask, error: Error) {
        DispatchQueue.main.async {
            self.progressView.isHidden = true
            self.statusLabel.text = "Download failed"
            self.errorLabel.text = self.getUserFriendlyErrorMessage(error)
            self.errorLabel.isHidden = false
        }
    }
    
    func getUserFriendlyErrorMessage(_ error: Error) -> String {
        let nsError = error as NSError
        
        switch nsError.code {
        case BIDownloadErrorCode.networkError.rawValue:
            return "Please check your internet connection and try again."
            
        case BIDownloadErrorCode.insufficientSpace.rawValue:
            return "Not enough storage space. Please free up some space."
            
        case BIDownloadErrorCode.authenticationError.rawValue:
            return "Authentication failed. Please login again."
            
        case BIDownloadErrorCode.serverError.rawValue:
            if let serverCode = nsError.userInfo["ServerCode"] as? Int {
                switch serverCode {
                case 404:
                    return "File not found on server."
                case 500:
                    return "Server error. Please try again later."
                case 503:
                    return "Service temporarily unavailable."
                default:
                    return "Server error (\(serverCode)). Please try again."
                }
            }
            return "Server error. Please try again later."
            
        default:
            return "Download failed. Please try again."
        }
    }
}

Error Logging

Debug Logging

// Enable detailed logging
var config = BIDownloadManagerConfiguration()
config.enableLogging = true
config.logLevel = .verbose
BIDownloadManager.shared.configure(config)

// Custom error logger
class ErrorLogger {
    static func logError(_ error: Error, task: BIDownloadTask) {
        let timestamp = DateFormatter.localizedString(from: Date(), dateStyle: .short, timeStyle: .long)
        let nsError = error as NSError
        
        var logMessage = """
        [\(timestamp)] Download Error
        Task ID: \(task.id)
        URL: \(task.url.absoluteString)
        Error Code: \(nsError.code)
        Error Domain: \(nsError.domain)
        Description: \(error.localizedDescription)
        """
        
        if let userInfo = nsError.userInfo as? [String: Any] {
            logMessage += "\nUser Info:"
            for (key, value) in userInfo {
                logMessage += "\n  \(key): \(value)"
            }
        }
        
        print(logMessage)
        
        // Save to file for crash reports
        saveToErrorLog(logMessage)
    }
    
    static func saveToErrorLog(_ message: String) {
        let documentsPath = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
        let logFile = documentsPath.appendingPathComponent("download_errors.log")
        
        let logEntry = "\n---\n\(message)\n"
        
        if let data = logEntry.data(using: .utf8) {
            if FileManager.default.fileExists(atPath: logFile.path) {
                if let fileHandle = try? FileHandle(forWritingTo: logFile) {
                    fileHandle.seekToEndOfFile()
                    fileHandle.write(data)
                    fileHandle.closeFile()
                }
            } else {
                try? data.write(to: logFile)
            }
        }
    }
}

// Usage
func downloadDidFail(_ task: BIDownloadTask, error: Error) {
    ErrorLogger.logError(error, task: task)
    // Handle error...
}

Recovery Strategies

Strategy Matrix

Error Type Strategy Action
Network Timeout Immediate Retry Retry 3 times with exponential backoff
No Internet Wait and Retry Monitor network, retry when available
Server 5xx Delayed Retry Wait 30s-5min, retry
Server 404 No Retry Show error, don't retry
Auth Failed Refresh Credentials Refresh token, retry with new credentials
Disk Full User Action Required Prompt user to free space
Invalid URL No Retry Fix URL, don't retry automatically

Implementation

func handleError(_ error: Error, task: BIDownloadTask) -> RecoveryAction {
    let nsError = error as NSError
    
    switch nsError.code {
    case BIDownloadErrorCode.networkError.rawValue:
        // Check specific network error
        if let underlying = nsError.userInfo[NSUnderlyingErrorKey] as? NSError {
            switch underlying.code {
            case NSURLErrorTimedOut:
                return .retryImmediately
            case NSURLErrorNotConnectedToInternet:
                return .waitForNetwork
            default:
                return .retryWithBackoff
            }
        }
        return .retryWithBackoff
        
    case BIDownloadErrorCode.serverError.rawValue:
        if let serverCode = nsError.userInfo["ServerCode"] as? Int {
            switch serverCode {
            case 500...599:
                return .retryDelayed
            case 404:
                return .noRetry("File not found")
            case 401, 403:
                return .refreshAuth
            default:
                return .noRetry("Server error")
            }
        }
        return .noRetry("Unknown server error")
        
    case BIDownloadErrorCode.authenticationError.rawValue:
        return .refreshAuth
        
    case BIDownloadErrorCode.insufficientSpace.rawValue:
        return .userActionRequired("Free up disk space")
        
    case BIDownloadErrorCode.invalidURL.rawValue:
        return .noRetry("Invalid URL")
        
    default:
        return .retryWithBackoff
    }
}

enum RecoveryAction {
    case retryImmediately
    case retryWithBackoff
    case retryDelayed
    case waitForNetwork
    case refreshAuth
    case userActionRequired(String)
    case noRetry(String)
}

Error Handling Best Practices

  • Always handle errors - Never ignore downloadDidFail callback
  • Provide user feedback - Show clear, actionable error messages
  • Log errors - Keep error logs for debugging and analytics
  • Implement retry logic - Use exponential backoff for network errors
  • Check preconditions - Validate disk space, network before starting
  • Handle authentication - Refresh tokens automatically when possible
  • Clean up resources - Delete partial files on failure
  • Monitor network - Detect network changes and adapt
  • Test edge cases - Simulate errors in testing
  • Graceful degradation - Fall back to single-threaded if multipart fails
Pro Tip: Use analytics to track error rates and identify patterns. This helps improve reliability and user experience.
