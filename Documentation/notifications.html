Notifications

Monitor downloads system-wide with NotificationCenter

Overview

BICurlDownloader posts notifications through NotificationCenter, allowing any part of your app to observe download events without direct delegate connections.

Use Case: Notifications are perfect for updating UI in multiple view controllers, analytics tracking, or coordinating with other app components.

Notification Types

Notification Name When Posted UserInfo Keys
managerStateDidChange Manager state changes activeTasksCount, queuedTasksCount
taskDidStart Download starts task
taskDidUpdateProgress Progress updates task, progress, bytesDownloaded, totalBytes
taskDidComplete Download completes task, filePath
taskDidFail Download fails task, error
taskDidPause Download paused task
taskDidResume Download resumed task
taskDidCancel Download cancelled task

Basic Usage

Swift Implementation

import BICurlDownloader

class DownloadObserver {
    
    init() {
        setupNotifications()
    }
    
    private func setupNotifications() {
        let center = NotificationCenter.default
        
        // Observe all download events
        center.addObserver(
            self,
            selector: #selector(downloadStarted(_:)),
            name: BIDownloadNotification.taskDidStart,
            object: nil
        )
        
        center.addObserver(
            self,
            selector: #selector(downloadProgress(_:)),
            name: BIDownloadNotification.taskDidUpdateProgress,
            object: nil
        )
        
        center.addObserver(
            self,
            selector: #selector(downloadCompleted(_:)),
            name: BIDownloadNotification.taskDidComplete,
            object: nil
        )
        
        center.addObserver(
            self,
            selector: #selector(downloadFailed(_:)),
            name: BIDownloadNotification.taskDidFail,
            object: nil
        )
    }
    
    @objc private func downloadStarted(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask else { return }
        print("Download started: \(task.url.lastPathComponent)")
    }
    
    @objc private func downloadProgress(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask,
              let progress = notification.userInfo?["progress"] as? Double else { return }
        
        let percentage = Int(progress * 100)
        print("\(task.url.lastPathComponent): \(percentage)%")
    }
    
    @objc private func downloadCompleted(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask,
              let filePath = notification.userInfo?["filePath"] as? String else { return }
        
        print("Completed: \(filePath)")
    }
    
    @objc private func downloadFailed(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask,
              let error = notification.userInfo?["error"] as? Error else { return }
        
        print("Failed: \(error.localizedDescription)")
    }
    
    deinit {
        NotificationCenter.default.removeObserver(self)
    }
}

Objective-C Implementation

@implementation DownloadObserver

- (instancetype)init {
    self = [super init];
    if (self) {
        [self setupNotifications];
    }
    return self;
}

- (void)setupNotifications {
    NSNotificationCenter *center = [NSNotificationCenter defaultCenter];
    
    [center addObserver:self
               selector:@selector(downloadStarted:)
                   name:BIDownloadNotificationTaskDidStart
                 object:nil];
    
    [center addObserver:self
               selector:@selector(downloadProgress:)
                   name:BIDownloadNotificationTaskDidUpdateProgress
                 object:nil];
    
    [center addObserver:self
               selector:@selector(downloadCompleted:)
                   name:BIDownloadNotificationTaskDidComplete
                 object:nil];
    
    [center addObserver:self
               selector:@selector(downloadFailed:)
                   name:BIDownloadNotificationTaskDidFail
                 object:nil];
}

- (void)downloadStarted:(NSNotification *)notification {
    BIDownloadTask *task = notification.userInfo[@"task"];
    NSLog(@"Download started: %@", task.url.lastPathComponent);
}

- (void)downloadProgress:(NSNotification *)notification {
    BIDownloadTask *task = notification.userInfo[@"task"];
    NSNumber *progressNumber = notification.userInfo[@"progress"];
    double progress = [progressNumber doubleValue];
    
    NSInteger percentage = (NSInteger)(progress * 100);
    NSLog(@"%@: %ld%%", task.url.lastPathComponent, (long)percentage);
}

- (void)downloadCompleted:(NSNotification *)notification {
    NSString *filePath = notification.userInfo[@"filePath"];
    NSLog(@"Completed: %@", filePath);
}

- (void)downloadFailed:(NSNotification *)notification {
    NSError *error = notification.userInfo[@"error"];
    NSLog(@"Failed: %@", error.localizedDescription);
}

- (void)dealloc {
    [[NSNotificationCenter defaultCenter] removeObserver:self];
}

@end

Manager State Notifications

Monitor Queue State

// Swift
class QueueMonitor {
    
    init() {
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(managerStateChanged(_:)),
            name: BIDownloadNotification.managerStateDidChange,
            object: nil
        )
    }
    
    @objc private func managerStateChanged(_ notification: Notification) {
        guard let activeCount = notification.userInfo?["activeTasksCount"] as? Int,
              let queuedCount = notification.userInfo?["queuedTasksCount"] as? Int else {
            return
        }
        
        print("Active downloads: \(activeCount)")
        print("Queued downloads: \(queuedCount)")
        print("Total: \(activeCount + queuedCount)")
        
        // Update badge or UI
        updateDownloadBadge(count: activeCount + queuedCount)
    }
    
    private func updateDownloadBadge(count: Int) {
        DispatchQueue.main.async {
            UIApplication.shared.applicationIconBadgeNumber = count
        }
    }
}

Objective-C Implementation

- (void)setupManagerObserver {
    [[NSNotificationCenter defaultCenter] addObserver:self
                                             selector:@selector(managerStateChanged:)
                                                 name:BIDownloadNotificationManagerStateDidChange
                                               object:nil];
}

- (void)managerStateChanged:(NSNotification *)notification {
    NSInteger activeCount = [notification.userInfo[@"activeTasksCount"] integerValue];
    NSInteger queuedCount = [notification.userInfo[@"queuedTasksCount"] integerValue];
    
    NSLog(@"Active: %ld, Queued: %ld", (long)activeCount, (long)queuedCount);
    
    [self updateDownloadBadge:activeCount + queuedCount];
}

Analytics Integration

Track Download Events

// Swift
class AnalyticsTracker {
    
    init() {
        setupNotifications()
    }
    
    private func setupNotifications() {
        let center = NotificationCenter.default
        
        center.addObserver(
            self,
            selector: #selector(trackDownloadStart(_:)),
            name: BIDownloadNotification.taskDidStart,
            object: nil
        )
        
        center.addObserver(
            self,
            selector: #selector(trackDownloadComplete(_:)),
            name: BIDownloadNotification.taskDidComplete,
            object: nil
        )
        
        center.addObserver(
            self,
            selector: #selector(trackDownloadFailure(_:)),
            name: BIDownloadNotification.taskDidFail,
            object: nil
        )
    }
    
    @objc private func trackDownloadStart(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask else {
            return
        }
        
        // Track to analytics service
        Analytics.logEvent("download_started", parameters: [
            "url": task.url.absoluteString,
            "file_type": task.url.pathExtension,
            "timestamp": Date().timeIntervalSince1970
        ])
    }
    
    @objc private func trackDownloadComplete(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask else {
            return
        }
        
        let duration = calculateDownloadDuration(task)
        let fileSize = task.progress.totalUnitCount
        
        Analytics.logEvent("download_completed", parameters: [
            "url": task.url.absoluteString,
            "file_size": fileSize,
            "duration": duration,
            "success": true
        ])
    }
    
    @objc private func trackDownloadFailure(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask,
              let error = notification.userInfo?["error"] as? NSError else {
            return
        }
        
        Analytics.logEvent("download_failed", parameters: [
            "url": task.url.absoluteString,
            "error_code": error.code,
            "error_message": error.localizedDescription
        ])
    }
    
    private func calculateDownloadDuration(_ task: BIDownloadTask) -> TimeInterval {
        // Implementation to calculate duration
        return 0
    }
}

Progress Throttling

Progress notifications are automatically throttled to avoid excessive updates.

Built-in Throttling

// Progress updates are throttled internally
// Maximum update frequency: ~100ms (10 updates per second)
// This prevents UI performance issues

// You receive progress updates at reasonable intervals automatically

Custom Throttling

// Swift - Additional throttling if needed
class ThrottledProgressObserver {
    private var lastUpdateTime: Date?
    private let minimumInterval: TimeInterval = 0.5 // 500ms
    
    @objc private func downloadProgress(_ notification: Notification) {
        let now = Date()
        
        // Check if enough time has passed
        if let lastUpdate = lastUpdateTime,
           now.timeIntervalSince(lastUpdate) < minimumInterval {
            return // Skip this update
        }
        
        lastUpdateTime = now
        
        // Process the update
        if let progress = notification.userInfo?["progress"] as? Double {
            updateUI(progress: progress)
        }
    }
}

Custom Notifications

Post Your Own Notifications

// Swift
// Define custom notification names
extension Notification.Name {
    static let downloadBatchCompleted = Notification.Name("downloadBatchCompleted")
    static let allDownloadsFinished = Notification.Name("allDownloadsFinished")
}

class DownloadCoordinator {
    private var activeDownloads: Set = []
    
    func startBatchDownload(urls: [URL]) {
        for url in urls {
            let task = BIDownloadManager.shared.download(
                url: url,
                destinationPath: "/path/to/destination"
            )
            activeDownloads.insert(task.id)
        }
    }
    
    @objc private func downloadCompleted(_ notification: Notification) {
        guard let task = notification.userInfo?["task"] as? BIDownloadTask else {
            return
        }
        
        activeDownloads.remove(task.id)
        
        // Check if batch is complete
        if activeDownloads.isEmpty {
            NotificationCenter.default.post(
                name: .downloadBatchCompleted,
                object: self,
                userInfo: ["totalDownloads": urls.count]
            )
        }
    }
}

Thread Safety

Important: Notifications may be posted from background threads. Always dispatch UI updates to the main thread.

Safe UI Updates

// Swift
@objc private func downloadProgress(_ notification: Notification) {
    guard let progress = notification.userInfo?["progress"] as? Double else {
        return
    }
    
    // ALWAYS update UI on main thread
    DispatchQueue.main.async {
        self.progressView.progress = Float(progress)
        self.statusLabel.text = "\(Int(progress * 100))%"
    }
}

// Objective-C
- (void)downloadProgress:(NSNotification *)notification {
    NSNumber *progressNumber = notification.userInfo[@"progress"];
    double progress = [progressNumber doubleValue];
    
    // ALWAYS update UI on main thread
    dispatch_async(dispatch_get_main_queue(), ^{
        self.progressView.progress = (float)progress;
        self.statusLabel.text = [NSString stringWithFormat:@"%ld%%", (long)(progress * 100)];
    });
}

Best Practices

  • Remove observers - Always remove observers in deinit or dealloc
  • Use main thread for UI - Dispatch all UI updates to main thread
  • Filter notifications - Check task ID or note to handle only relevant events
  • Avoid heavy processing - Keep notification handlers lightweight
  • Consider throttling - Add custom throttling for progress if needed
  • Handle missing data - Always safely unwrap userInfo values
  • Use weak references - Avoid retain cycles in closures
  • Document custom notifications - If posting custom events

Memory Management Example

// Swift
class DownloadViewController: UIViewController {
    
    override func viewDidLoad() {
        super.viewDidLoad()
        setupNotifications()
    }
    
    private func setupNotifications() {
        NotificationCenter.default.addObserver(
            self,
            selector: #selector(downloadProgress(_:)),
            name: BIDownloadNotification.taskDidUpdateProgress,
            object: nil
        )
    }
    
    @objc private func downloadProgress(_ notification: Notification) {
        // Use [weak self] in async closures
        DispatchQueue.main.async { [weak self] in
            guard let self = self else { return }
            self.updateProgressUI()
        }
    }
    
    deinit {
        // Critical: remove observers
        NotificationCenter.default.removeObserver(self)
    }
}
